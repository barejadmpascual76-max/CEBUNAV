def create_account():
    print("\n==========================================")
    print("         CREATE YOUR CEBUNAV ACCOUNT ")
    print("==========================================")

    # Personal Information
    name = input("Enter your full name: ").strip()
    username = input("Create a username: ").strip()
    
    while True:
        email_address = input("Email address: ").strip().lower()
        if "@" in email_address and ".com" in email_address:
            break
        else:
             print("Invalid Email Address (Must contain '@' and '.com')")

    # Password
    while True:
        password = input("Create your password (8 characters): ")

        if len(password) != 8:
            print("Password must be exactly 8 characters.")
            continue

        confirm_password = input("Confirm your password: ")

        if password == confirm_password:
            print("Password created successfully!")
            break
        else:
            print("Passwords do not match.")

    accounts[username] = {
        "name": name,
        "email_address": email_address,
        "password": password,
    }

    # Account Created
    print("\n==========================================")
    print("       CEBUNAV ACCOUNT CREATED!")
    print("==========================================")
    print("Welcome to CEBUNAV, " + name + "!")
    print("Your account has been saved.")
    print("You can now log in.")
    print("==========================================")

    input("\nPress Enter to return to the ACCOUNT CREATION...")

def account_type():
    print("\n1. Regular Passenger")
    print("2. Student")
    print("3. Senior Citizen")
    print("4. PWD")

    types = {
        "1": "regular",
        "2": "student",
        "3": "senior",
        "4": "pwd"
    }

    choice = input("Choose account type: ")
    return types[choice] if choice in types else account_type()
  
def calculate_fare(fare, account):
    discount = 0.20 if account.lower() in ["student", "senior", "pwd"] else 0
    final_fare = fare * (1 - discount)

    print(f"Original Fare: ₱{fare:.2f}")
    print(f"Discount: {discount:.0%}")
    print(f"Final Fare: ₱{final_fare:.2f}")

    return final_fare
  

def route_finder(account):
    print("\n========== FIND YOUR ROUTE ==========")
    places = []
    for stops in jeeps.values():
        for stop in stops: 
          if stop not in places: 
             places.append(stop)
            
    print("Places Available: ")
    for places in places: 
       print(" -- " + places.title())
       

    start = input("Enter your location: ").lower().strip()
    destination = input("Enter your destination: ").lower().strip()

    found = False

    print("\n========== AVAILABLE JEEPNEYS ==========")

    for code, stops in jeeps.items():
        if start in stops and destination in stops:
            found = True

            print(f"\nJeepney Code: {code}")
            print("Route:", " --> ".join(stops))

    if not found:
        print("Sorry, no jeepney found for this route.")
        return

    # Calculate fare
    fare = 0

    if start == "urgello":
        if destination == "emall":
            fare = 15
        elif destination == "colon":
            fare = 13
        elif destination == "sm city":
            fare = 17
        elif destination == "parkmall":
            fare = 20

    if fare > 0:
        print("\n========== FARE DETAILS ==========")
        calculate_fare(fare, account)
    else:
        print("\nFare for this route is not yet available.")

    print("======================================")

def dashboard(username=None, account="Regular Passenger"):
    while True:
        print("\n================= CEBUNAV DASHBOARD =====================")
        print("1. Find Route")
        print("2. Handbook")

        if username is None:
            print("3. Create Account\n")
        else:
            print("3. Log out\n")

        choice = input("Choose 1, 2, or 3: ").strip()

        if choice == "1":
            route_finder(account)

        elif choice == "2":
            handbook()

        elif choice == "3":
            if username is None:
                print("Create Account!")
                create_account()
            else:
                print("Logged out. Amping!")
            break

        else:
            print("Invalid choice.")
